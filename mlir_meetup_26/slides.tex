\begin{frame}[fragile]{Can we do better? Online Softmax}
  \lstset{basicstyle=\ttfamily\small}
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.44\textwidth}
      \[
        \begin{aligned}
          m &= \max_j x_j \\
          s &= \sum_j e^{\,x_j - m} \\
          \mathrm{softmax}(x)_j &= \frac{e^{\,x_j - m}}{s}
        \end{aligned}
      \]
      \vspace{2mm}
      \[
        e^{\,x_j - m_{\text{new}}} = e^{\,x_j - m_{\text{old}}} \cdot
          \textcolor{inteltitle}{e^{\,m_{\text{old}} - m_{\text{new}}}}
      \]
    \end{column}
    \begin{column}{0.53\textwidth}
\begin{lstlisting}
m = -inf;  s = 0
for each tile x_t of x:
|p  m_new = max(m, max(x_t))|
|e  s = s |* |cexp(m - m_new)|    // correction
|e      + sum(exp(x_t - m_new))|
  m = m_new
\end{lstlisting}
    \end{column}
  \end{columns}

  \takeaway{One pass over \texttt{x}: rescale the running sum whenever the max
    changes}
\end{frame}

\begin{frame}{When/Why does this work?}
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.30\textwidth}
      \footnotesize
      \begin{align*}
        m &= \max_j x_j \\
        \mathrm{softmax}(x)_j &= \frac{e^{\,x_j - m}}{s} \\
        s &= \sum_j e^{\,x_j - m} \\
          &= \sum_j \underbrace{e^{\,x_j}}_{\text{data}} \cdot
             \underbrace{\gfac{m}}_{\text{R1}}
      \end{align*}
    \end{column}
    \begin{column}{0.68\textwidth}
      \small
      \setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{2pt}%
      \setlength{\jot}{2pt}
      \textbf{Stage $i-1$}
      \begin{align}
        s_{i-1} &= \textstyle\sum_{j<i} e^{\,x_j - m_{i-1}} \\
                &= e^{\,x_1}\gfac{m_{i-1}} + e^{\,x_2}\gfac{m_{i-1}} + \dots
                   + e^{\,x_{i-1}}\gfac{m_{i-1}}
      \end{align}
      \textbf{Stage $i$}
      \begin{align}
        m_i &= \max(m_{i-1}, x_i) \\
        s_{i-1} \cdot \frac{1}{\gfac{m_{i-1}}} &= e^{\,x_1} + e^{\,x_2} + \dots
          + e^{\,x_{i-1}} \\
        s_{i-1} \cdot
          \tikz[remember picture, baseline=(cf.base)]
            \node[draw=inteltitle, line width=0.8pt, rounded corners=1pt,
                  inner sep=1pt] (cf)
              {$\displaystyle\frac{\gfac{m_i}}{\gfac{m_{i-1}}}$};
          &= e^{\,x_1 - m_i}
          + e^{\,x_2 - m_i} + \dots + e^{\,x_{i-1} - m_i} \\
        s_i &= s_{i-1} \cdot e^{\,m_{i-1} - m_i} + e^{\,x_i - m_i}
      \end{align}
      \begin{tikzpicture}[remember picture, overlay]
        \node[anchor=east, font=\scriptsize, text=inteltitle] (cflbl)
          at ([xshift=-10mm, yshift=-7mm] cf.south west) {correction factor};
        \draw[-{Latex[length=1.8mm]}, inteltitle, line width=0.8pt]
          (cflbl.east) to[out=0, in=-90] (cf.south);
      \end{tikzpicture}
    \end{column}
  \end{columns}

  \takeaway{R2 is a \emph{sum}, and each summed term is $f(x) \cdot g(\mathrm{R1})$
    (multiplicatively separable)\\
    Multiplication distributes over addition $\Rightarrow$ one rescale corrects
    every term}
\end{frame}

\begin{frame}{Legality Conditions}
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.56\textwidth}
      For a reduction chain R1 $\rightarrow$ E $\rightarrow$ R2:
      \vspace{2mm}
      \begin{enumerate}
        \setlength{\itemsep}{3mm}
        \item \textbf{R1} can be \emph{any} reduction
        \item \textbf{E} (elementwise) uses R1's result \emph{broadcast} along
          R1's reduction axis
        \item E's use of R1 is \emph{multiplicatively separable}:
          $E(x, r) = f(x) \cdot g(r)$
        \item \textbf{R2} is a \emph{sum} reduction that reduces over the same
          axis as R1
      \end{enumerate}
    \end{column}
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[
          box/.style={draw, rounded corners=3pt, minimum width=30mm,
                      minimum height=9mm, align=center, font=\small},

\begin{frame}[fragile]{Multiplicative Separability}
  A lightweight \emph{dataflow analysis} over E's body proves
  $E(x, m) = f(x) \cdot g(m)$

  \vspace{2mm}
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.44\textwidth}
      \hspace*{5mm}%
      \begin{minipage}{\dimexpr\linewidth-5mm\relax}
      \small
      \textbf{Facts}, propagated from inputs to yield
      \begin{itemize}
        \setlength{\itemsep}{0mm}
        \item \dfact{IND}: independent of R1
        \item \dfact{ADD}: R1 enters \emph{additively}
        \item \dfact{MUL}: R1 enters \emph{multiplicatively}
      \end{itemize}
      \vspace{0.5mm}
      \textbf{Transfer rules}\\[0.5mm]
      \renewcommand{\arraystretch}{0.95}%
      \begin{tabular}{@{}ll@{}}
        \texttt{add}, \texttt{sub} & keep \dfact{ADD} \\
        \texttt{mul}, \texttt{div} & keep \dfact{MUL} \\
        \texttt{exp}               & \dfact{ADD} $\rightarrow$ \dfact{MUL} \\
        \texttt{neg} / \texttt{abs} & keep all / keep \dfact{MUL} \\
        other ops                  & no facts
      \end{tabular}
      \end{minipage}
    \end{column}
    \begin{column}{0.54\textwidth}
      \small
      \textbf{Example: softmax}, $E = \exp(x - m)$\\[1mm]
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize]
// %x is {IND}, %m is {ADD, MUL}
%d = arith.subf %x, %m   // {ADD}
%e = math.exp %d         // {MUL}
linalg.yield %e          // {MUL}
\end{lstlisting}
      \vspace{1mm}
      \colorbox{pro!15}{%
        \parbox{\dimexpr\linewidth-2\fboxsep\relax}{\centering\vspace{1mm}
          Yield is \dfact{MUL} and depends on \texttt{\%m}
          $\Rightarrow$ separable \textcolor{pro}{\checkmark}\vspace{1mm}}}
    \end{column}
  \end{columns}

  \takeaway{Accept E only if its yield is \dfact{MUL} and depends on R1}
\end{frame}
