\begin{frame}[fragile]{Can we do better? Online Softmax}
  \lstset{basicstyle=\ttfamily\small}
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.44\textwidth}
      \[
        \begin{aligned}
          m &= \max_j x_j \\
          s &= \sum_j e^{\,x_j - m} \\
          \mathrm{softmax}(x)_j &= \frac{e^{\,x_j - m}}{s}
        \end{aligned}
      \]
      \vspace{2mm}
      \[
        e^{\,x_j - m_{\text{new}}} = e^{\,x_j - m_{\text{old}}} \cdot
          \textcolor{inteltitle}{e^{\,m_{\text{old}} - m_{\text{new}}}}
      \]
    \end{column}
    \begin{column}{0.53\textwidth}
\begin{lstlisting}
m = -inf;  s = 0
for each tile x_t of x:
|p  m_new = max(m, max(x_t))|
|e  s = s |* |cexp(m - m_new)|    // correction
|e      + sum(exp(x_t - m_new))|
  m = m_new
\end{lstlisting}
    \end{column}
  \end{columns}

  \takeaway{One pass over \texttt{x}: rescale the running sum whenever the max
    changes}
\end{frame}

\begin{frame}{When/Why does this work?}
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.30\textwidth}
      \footnotesize
      \begin{align*}
        m &= \max_j x_j \\
        \mathrm{softmax}(x)_j &= \frac{e^{\,x_j - m}}{s} \\
        s &= \sum_j e^{\,x_j - m} \\
          &= \sum_j \underbrace{e^{\,x_j}}_{\text{data}} \cdot
             \underbrace{\gfac{m}}_{\text{R1}}
      \end{align*}
    \end{column}
    \begin{column}{0.68\textwidth}
      \small
      \setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{2pt}%
      \setlength{\jot}{2pt}
      \textbf{Stage $i-1$}
      \begin{align}
        s_{i-1} &= \textstyle\sum_{j<i} e^{\,x_j - m_{i-1}} \\
                &= e^{\,x_1}\gfac{m_{i-1}} + e^{\,x_2}\gfac{m_{i-1}} + \dots
                   + e^{\,x_{i-1}}\gfac{m_{i-1}}
      \end{align}
      \textbf{Stage $i$}
      \begin{align}
        m_i &= \max(m_{i-1}, x_i) \\
        s_{i-1} \cdot \frac{1}{\gfac{m_{i-1}}} &= e^{\,x_1} + e^{\,x_2} + \dots
          + e^{\,x_{i-1}} \\
        s_{i-1} \cdot
          \tikz[remember picture, baseline=(cf.base)]
            \node[draw=inteltitle, line width=0.8pt, rounded corners=1pt,
                  inner sep=1pt] (cf)
              {$\displaystyle\frac{\gfac{m_i}}{\gfac{m_{i-1}}}$};
          &= e^{\,x_1 - m_i}
          + e^{\,x_2 - m_i} + \dots + e^{\,x_{i-1} - m_i} \\
        s_i &= s_{i-1} \cdot e^{\,m_{i-1} - m_i} + e^{\,x_i - m_i}
      \end{align}
      \begin{tikzpicture}[remember picture, overlay]
        \node[anchor=east, font=\scriptsize, text=inteltitle] (cflbl)
          at ([xshift=-10mm, yshift=-7mm] cf.south west) {correction factor};
        \draw[-{Latex[length=1.8mm]}, inteltitle, line width=0.8pt]
          (cflbl.east) to[out=0, in=-90] (cf.south);
      \end{tikzpicture}
    \end{column}
  \end{columns}

  \takeaway{R2 is a \emph{sum}, and each summed term is $f(x) \cdot g(r_1)$
    (multiplicatively separable)\\
    Multiplication distributes over addition $\Rightarrow$ one rescale corrects
    every term}
\end{frame}

\begin{frame}{Legality Conditions}
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.56\textwidth}
      For a reduction chain R1 $\rightarrow$ E $\rightarrow$ R2:
      \vspace{2mm}
      \begin{enumerate}
        \setlength{\itemsep}{3mm}
        \item \textbf{R1} can be \emph{any} reduction
        \item \textbf{E} (elementwise) uses R1's result \emph{broadcast} along

        \item E's use of R1 is \emph{multiplicatively separable}:
          $E(x, r_1) = f(x) \cdot g(r_1)$
        \item \textbf{R2} is a \emph{sum} reduction that reduces over the same
          axis as R1
      \end{enumerate}
    \end{column}

\begin{frame}[fragile]{Multiplicative Separability}
  A lightweight \emph{dataflow analysis} over E's body proves
  $E(x, r_1) = f(x) \cdot g(r_1)$

  \vspace{2mm}
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.44\textwidth}
      \hspace*{5mm}%
      \begin{minipage}{\dimexpr\linewidth-5mm\relax}
      \small
      \textbf{Facts}, propagated from inputs to yield
      \begin{itemize}
        \setlength{\itemsep}{0mm}
        \item \dfact{IND}: independent of R1
        \item \dfact{ADD}: R1 enters \emph{additively}
        \item \dfact{MUL}: R1 enters \emph{multiplicatively}
      \end{itemize}
      \vspace{0.5mm}
      \textbf{Transfer rules}\\[0.5mm]
      \renewcommand{\arraystretch}{0.95}%
      \begin{tabular}{@{}ll@{}}
        \texttt{add}, \texttt{sub} & keep \dfact{ADD} \\
        \texttt{mul}, \texttt{div} & keep \dfact{MUL} \\
        \texttt{exp}               & \dfact{ADD} $\rightarrow$ \dfact{MUL} \\
        \texttt{neg} / \texttt{abs} & keep all / keep \dfact{MUL} \\
        other ops                  & no facts
      \end{tabular}
      \end{minipage}
    \end{column}
    \begin{column}{0.54\textwidth}
      \small
      \textbf{Example: softmax}, $E = \exp(x - m)$\\[1mm]
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize]
// %x is {IND}, %m is {ADD, MUL}
%d = arith.subf %x, %m   // {ADD}
%e = math.exp %d         // {MUL}
linalg.yield %e          // {MUL}
\end{lstlisting}
      \vspace{1mm}
      \colorbox{pro!15}{%
        \parbox{\dimexpr\linewidth-2\fboxsep\relax}{\centering\vspace{1mm}
          Yield is \dfact{MUL} and depends on \texttt{\%m}
          $\Rightarrow$ separable \textcolor{pro}{\checkmark}\vspace{1mm}}}
    \end{column}
  \end{columns}

  \takeaway{Accept E only if its yield is \dfact{MUL} and depends on R1}
\end{frame}

\begin{frame}[fragile]{Deriving Online Correction}
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.46\textwidth}
      \small
      \begin{enumerate}
        \setlength{\itemsep}{2mm}
        \item Separability gives $E(x, r_1) = f(x) \cdot g(r_1)$. To isolate
          $r_1$'s effect, build $E'(r_1) = g(r_1)$ by making $f(x) = 1$
        \item Compute the \emph{forward slice} of $r_1$ in E's body
        \item Set every slice operand that does not depend on $r_1$ to the op's
          \emph{neutral element} (0 for \texttt{add}/\texttt{sub}, 1 for
          \texttt{mul}/\texttt{div})
        \item Evaluate $E'$ at $r_1^{\text{old}}$ and $r_1^{\text{new}}$:
          correction $= E'(r_1^{\text{new}}) / E'(r_1^{\text{old}})$
      \end{enumerate}
    \end{column}
    \begin{column}{0.50\textwidth}
      \small
      \textbf{Example:} $E(x, r_1) = \exp(x - r_1)$\\[1mm]
      \lstset{basicstyle=\ttfamily\fontsize{7.5}{9}\selectfont}
      \begin{minipage}[t]{0.45\linewidth}
\begin{lstlisting}
// E

%d = arith.subf %x, %r1
%e = math.exp %d
linalg.yield %e
\end{lstlisting}
      \end{minipage}\hfill
      \begin{minipage}[t]{0.54\linewidth}
\begin{lstlisting}
// E': %x := 0.0
|c%c0 = arith.constant 0.0|
%d = arith.subf |c%c0|, %r1
%e = math.exp %d
linalg.yield %e
\end{lstlisting}
      \end{minipage}

      \vspace{2mm}
      \[
        E'(r_1) = \exp(0.0 - r_1)
      \]
      \colorbox{pro!15}{%
        \parbox{\dimexpr\linewidth-2\fboxsep\relax}{\centering\vspace{1mm}
          correction $= \dfrac{\exp(0.0 - r_1^{\text{new}})}{\exp(0.0 - r_1^{\text{old}})}$
          \vspace{1mm}}}
    \end{column}
  \end{columns}
\end{frame}

% =============================================================== Proposed op
\begin{frame}[fragile]{Proposal: \textit{structured.fuse\_dependent\_reduction\_ops}}
  Three operand handles, one result:
  \begin{enumerate}\small
    \item \textbf{\code{R1}} --- already tiled producer reduction
      (\code{scf.for} from \code{tile\_using\_for} with \code{FullReduction},
      accumulator carried as \code{iter\_arg}; may carry several results).
    \item \textbf{\code{E}} --- all-parallel elementwise op consuming an \code{R1}
      result as a broadcast input.
    \item \textbf{\code{R2}} --- still-untiled consumer reduction over the same
      axis, reducing \code{E}'s result.
  \end{enumerate}
\begin{lstlisting}
%loop = ..... -> !transform.any_op   // R1: the already tiled reduction loop
%e    = ..... -> !transform.any_op   // E : the elementwise term
%r2   = ..... -> !transform.any_op   // R2: the consumer reduction

%fused = transform.structured.fuse_dependent_reduction_ops %e, %r2 into %loop
    : (!transform.any_op, !transform.any_op, !transform.any_op)
    -> !transform.any_op
\end{lstlisting}
  \takeaway{Returns the fused loop, now carrying \code{E} and \code{R2}\\
    The op can be applied repeatedly to fuse chains of dependent reductions}
\end{frame}

    \node[st, right=6mm of s2] (s3) {Online correction};
    \node[font=\scriptsize\bfseries, text=pro!70!black, anchor=north]
      (lbl) at ([yshift=-1.5mm] $(s2.south)!0.5!(s3.south)$) {Proposed transformation};
    \draw[arr] (s1) -- (s2);
    \draw[arr] (s2) -- (s3);
    \begin{scope}[on background layer]
      \node[fill=pro!12, rounded corners=4pt, inner sep=2mm,
            fit=(s2)(s3)(lbl)] {};
    \end{scope}
  \end{tikzpicture}

  \vspace{4mm}
  \raggedright
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.40\textwidth}
      \footnotesize
      \begin{itemize}
        \setlength{\itemsep}{2mm}
        \item \code{E} is fused first (\code{scf::tileAndFuseConsumer}), then
          immediately re-tiled to match \code{R1}'s tile
        \item \code{R2} is fused against the slice that \code{E} yields
        \item Implemented as an \code{OpRewritePattern} on \code{R2}
      \end{itemize}
      \vspace{12mm}  % lift the points above the code column's center
    \end{column}
    \begin{column}{0.56\textwidth}
      \lstset{basicstyle=\ttfamily\fontsize{7.5}{9}\selectfont}
      {\scriptsize\textbf{Input:} softmax, max already tiled}
\begin{lstlisting}
|pfor j in tiles(N):|              // R1
|p  m = max(m, max(x[j]))|
p = exp(x - m)                  // E
|es = sum(p)|                      // R2
\end{lstlisting}
      \vspace{1mm}
      {\scriptsize\textbf{After fusion:} one loop}
\begin{lstlisting}
for j in tiles(N):
|p  m_new = max(m, max(x[j]))|     // R1
  p     = exp(x[j] - m_new)     // E, re-tiled
|c  s     = s * exp(m - m_new)|    // online correction
|e  s     = s + sum(p)|            // R2, re-tiled
  m     = m_new
\end{lstlisting}
    \end{column}
  \end{columns}
\end{frame}

% =============================================================== Example
\begin{frame}{Example: Standard to Fused Attention}
  \begin{itemize}\small
    \item \code{x} is the score matrix: \code{x = Q @ K\textsuperscript{T}}
    \item All reductions are over the inner \code{j} axis
    \item Normalization is deferred (flash-attention style)
  \end{itemize}
  \[
  \begin{aligned}
    R1 \;(\max):&\quad m[i]    = \max_j x[i,j] \\
    E \;(\text{term}):&\quad p[i,j]  = \exp(x[i,j] - m[i]) \\
    R2a \;(\text{sum}):&\quad s[i]    = \textstyle\sum_j p[i,j] \\
    R2b \;(P@V):&\quad o'[i,d] = \textstyle\sum_j p[i,j]\, V[j,d] \\
    \text{div}:&\quad o[i,d]  = o'[i,d] / s[i]
  \end{aligned}
  \]
  \takeaway{A single \code{E} feeds two consumer reductions $\Rightarrow$ requires
    \alert{2} applications of reduction fusion}
\end{frame}

\begin{frame}[fragile]{Input IR}
\begin{lstlisting}[escapeinside={(*}{*)}]
%m = scf.for %iv = 0 to 512 step 32                    // (*\textcolor{labelred}{\cbold{R1: row max}}*), already tiled
       iter_args(%acc_m = %ninf_64) {
  %xt   = tensor.extract_slice %x[0, %iv] [64, 32]     // score tile
  %mold = tensor.extract_slice %acc_m[0] [64]
  %mt   = linalg.generic ins(%xt) outs(%mold)          // max(x_tile, acc_m)
  %im   = tensor.insert_slice %mt into %acc_m[0] [64]
  scf.yield %im
}
%p = linalg.generic ins(%x, %m) outs(%empty_64x512)    // (*\textcolor{labelred}{\cbold{E: exp(x - m)}}*), full [64, 512]
%s = linalg.generic ins(%p) outs(%zero_64)             // (*\textcolor{labelred}{\cbold{R2a: row sum}}*), sum_j p
%o = linalg.generic ins(%p, %V) outs(%zero_64x64)      // (*\textcolor{labelred}{\cbold{R2b: P @ V}}*), sum_j p * V
%out = linalg.generic ins(%o, %s) outs(%empty_64x64)   // normalize: o / s
\end{lstlisting}
  \takeaway{Attention input shapes: \code{Q @ K\textsuperscript{T}} (64$\times$512),
    \code{K}, \code{V} (512$\times$64)}
\end{frame}

\begin{frame}[fragile]{The Transform Schedule}
\begin{lstlisting}[escapeinside={(*}{*)}]
%gen = transform.structured.match ops{["linalg.generic"]} in %arg0 .....
%r1, %e, %r2a, %r2b, %div = transform.split_handle %gen .....    // R1, E, R2a (sum), R2b (P @ V), div
%loop = transform.structured.match ops{["scf.for"]} in %arg0 .....

%fa = transform.structured.fuse_dependent_reduction_ops          // (*\textcolor{labelred}{\cbold{Fuse chain 1: max -> E -> row sum}}*)
        %e, %r2a into %loop .....

%e2 = transform.get_producer_of_operand %r2b[0] .....            // re-derive E (handle consumed)
%fb = transform.structured.fuse_dependent_reduction_ops          // (*\textcolor{labelred}{\cbold{Capture fused loop \%fa and fuse}}*)
        %e2, %r2b into %fa .....                                 // (*\textcolor{labelred}{\cbold{chain 2: max + sum -> E -> P @ V}}*)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Step 1: \textit{E} Fused --- Still at Full Extent}
\begin{lstlisting}[escapeinside={(*}{*)}]
%r:2 = scf.for %iv = 0 to 512 step 32                  // new iter_arg %acc_p: E's full result
         iter_args(%acc_m = %ninf_64, %acc_p = %empty_64x512) {
  %xt   = tensor.extract_slice %x[0, %iv] [64, 32]
  %mold = tensor.extract_slice %acc_m[0] [64]
  %mt   = linalg.generic ins(%xt) outs(%mold)          // max(x_tile, acc_m)
(*\hlmark{hlxf}*)  %xf   = tensor.extract_slice %x[0, 0] [64, 512]      // (*\textcolor{labelred}{\cbold{Fused E clone reads all the cols along the}}*)
  %pdst = tensor.extract_slice %acc_p[0, 0] [64, 512]  // (*\textcolor{labelred}{\cbold{reduction dim (512 cols) and writes}}*)
(*\hlmark{hlpt}*)  %pt   = linalg.generic ins(%xf, %mt) outs(%pdst)     // (*\textcolor{labelred}{\cbold{exp(x - m\_new) along the FULL row}}*)
  %im   = tensor.insert_slice %mt into %acc_m[0] [64]
(*\hlmark{hlip}*)  %ip   = tensor.insert_slice %pt into %acc_p[0, 0] [64, 512]
  scf.yield %im, %ip
}                                                      // outside: row sum, original E, P @ V, div
\end{lstlisting}
  \hlpaint{\hlrows{hlxf}{hlpt}\hlrows{hlip}{hlip}}
\end{frame}

\begin{frame}[fragile]{Step 2: \textit{E} Re-Tiled to the Reduction Tile}
\begin{lstlisting}[escapeinside={(*}{*)}]
%r:2 = scf.for %iv = 0 to 512 step 32                   // (*\textcolor{labelred}{\cbold{Reduction tile size = 32}}*)
         iter_args(%acc_m = %ninf_64, %acc_p = %empty_64x512) {
  %xt   = tensor.extract_slice %x[0, %iv] [64, 32]
  %mold = tensor.extract_slice %acc_m[0] [64]
  %mt   = linalg.generic ins(%xt) outs(%mold)           // max(x_tile, acc_m)
(*\hlmark{rtpdst}*)  %pdst = tensor.extract_slice %acc_p[0, %iv] [64, 32]  // (*\textcolor{labelred}{\cbold{E cut to the [64, 32] tile}}*)
(*\hlmark{rtpt}*)  %pt   = linalg.generic ins(%xt, %mt) outs(%pdst)      // (*\textcolor{labelred}{\cbold{exp(x\_tile - m\_new)}}*)
  %im   = tensor.insert_slice %mt into %acc_m[0] [64]
(*\hlmark{rtip}*)  %ip   = tensor.insert_slice %pt into %acc_p[0, %iv] [64, 32]
  scf.yield %im, %ip
}
\end{lstlisting}
  \hlpaint{\hlrows{rtpdst}{rtpt}\hlrows{rtip}{rtip}}
\end{frame}

\begin{frame}[fragile]{Step 3: The Row Sum Fused (and Re-Tiled)}
\begin{lstlisting}[escapeinside={(*}{*)}]
%r:3 = scf.for %iv = 0 to 512 step 32
         iter_args(%acc_m = %ninf_64, %acc_p = %empty_64x512, %acc_s = %zero_64) {
  %xt   = tensor.extract_slice %x[0, %iv] [64, 32]
  %mold = tensor.extract_slice %acc_m[0] [64]
  %mt   = linalg.generic ins(%xt) outs(%mold)       // max(x_tile, acc_m)
  %pdst = tensor.extract_slice %acc_p[0, %iv] [64, 32]
  %pt   = linalg.generic ins(%xt, %mt) outs(%pdst)  // exp(x_tile - m_new)
(*\hlmark{rssold}*)  %sold = tensor.extract_slice %acc_s[0] [64]       // (*\textcolor{labelred}{\cbold{Row sum fused against the  E tile}}*)
(*\hlmark{rsst}*)  %st   = linalg.generic ins(%pt) outs(%sold)       // (*\textcolor{labelred}{\cbold{(acc\_s + sum\_j p\_tile)}}*)
  ... yields: %mt, %pt, %st
}
\end{lstlisting}
  \hlpaint{\hlrows{rssold}{rsst}}
  \takeaway{Sum's only input was \code{E}, which is already re-tiled. So
    re-tiling is a no-op in this case}
\end{frame}

\begin{frame}[fragile]{Step 4: Online Correction --- Chain 1 Complete}
\begin{lstlisting}[escapeinside={(*}{*)}]
%r:3 = scf.for %iv = 0 to 512 step 32
         iter_args(%acc_m = %ninf_64, %acc_p = %empty_64x512, %acc_s = %zero_64) {
  %mt   = linalg.generic ins(%xt) outs(%mold)         // max(x_tile, acc_m)
  %pt   = linalg.generic ins(%xt, %mt) outs(%pdst)    // exp(x_tile - m_new)
  %sold = tensor.extract_slice %acc_s[0] [64]
(*\hlmark{octnew}*)  %tnew = linalg.generic ins(%mt) outs(%empty_64)     // (*\textcolor{labelred}{\cbold{Online correction: E'(m\_new) = exp(0.0 - m\_new)}}*)
  %told = linalg.generic ins(%mold) outs(%empty_64)   // (*\textcolor{labelred}{\cbold{E'(m\_old) = exp(0.0 - m\_old)}}*)
  %f    = linalg.elementwise <div> ins(%tnew, %told)  // (*\textcolor{labelred}{\cbold{correction factor = E'(m\_new)/E'(m\_old)}}*)
(*\hlmark{ocsc}*)  %sc   = linalg.elementwise <mul> ins(%sold, %f)     // (*\textcolor{labelred}{\cbold{acc\_s\_rescaled = acc\_s \char42{} factor}}*)
  %st   = linalg.generic ins(%pt) outs(%sc)           // sc + sum_j p_tile
  ... yields: %mt, %pt, %st
}
\end{lstlisting}
  \hlpaint{\hlrows{octnew}{ocsc}}
  \takeaway{Loop is again a tiled reduction loop (now returning two reductions)
    $\Rightarrow$ it can be the producer of the next chain}
\end{frame}

\begin{frame}[fragile]{Step 5: P @ V Fusion}
\begin{lstlisting}[escapeinside={(*}{*)}]
%r:5 = scf.for %iv = 0 to 512 step 32
         iter_args(%acc_m = %ninf_64, %acc_p1 = %empty_64x512, %acc_s = %zero_64,
                   %acc_p2 = %empty_64x512, %acc_o = %zero_64x64) {
  // %mt (max), %st (sum) fused in previous step
(*\hlmark{pvvt}*)  %vt    = tensor.extract_slice %V[%iv, 0] [32, 64]        // (*\textcolor{labelred}{\cbold{V slice is re-tiled}}*)
  %oold  = tensor.extract_slice %acc_o[0, 0] [64, 64]      // (*\textcolor{labelred}{\cbold{Online correction for P @ V acc}}*)
(*\hlmark{pvtnew}*)  %tnew2 = linalg.generic ins(%mt) outs(%empty_64x64)      // (*\textcolor{labelred}{\cbold{E'(m\_new) = exp(0.0 - m\_new)}}*)
  %told2 = linalg.generic ins(%mold) outs(%empty_64x64)    // (*\textcolor{labelred}{\cbold{E'(m\_old) = exp(0.0 - m\_old)}}*)
  %f2    = linalg.elementwise <div> ins(%tnew2, %told2)    // (*\textcolor{labelred}{\cbold{correction factor = E'(m\_new)/E'(m\_old)}}*)
(*\hlmark{pvoc}*)  %oc    = linalg.elementwise <mul> ins(%oold, %f2)        // (*\textcolor{labelred}{\cbold{acc\_o\_rescaled = acc\_o \char42{} factor}}*)
  %ot    = linalg.generic ins(%p2, %vt) outs(%oc)          // oc + sum_j p2 * V_tile
  // insert result tensors into the running accumulators
  ... yields: %mt, %p1, %st, %p2, %ot
}
%out = linalg.generic ins(%r#4, %r#2) outs(%empty_64x64)   // only the divide is left: o / s
\end{lstlisting}
  \hlpaint{\hlrows{pvvt}{pvvt}\hlrows{pvtnew}{pvoc}}
  \takeaway{\code{V} must be re-tiled (unlike before) because it's an extra input
    for R2\\
    Result is a fused online loop with max, sum and \code{P @ V} accumulators}
\end{frame}

\begin{frame}{Evaluation}
  \textbf{Hardware}
  \begin{itemize}
    \item Intel Battlemage (BMG) GPU
  \end{itemize}
  \vspace{3mm}
  \textbf{Attention kernel}
  \begin{itemize}
    \item \textbf{Input IR}: standard attention kernel written in Linalg,
      lowered for end-to-end execution using the upstream \alert{XeGPU} pipeline
    \item \textbf{Sizes}: \code{Z = 2} (batch), \code{H = 8} heads, head dim
      \code{= 64}, \code{f16}; \code{n\_ctx} $\in \{1024, 4096, 8192, 16384\}$
    \item \textbf{Baseline (unfused)}: same kernel tiled/fused with existing
      upstream transforms only --- no fusion on the reduction dim, score matrix
      recomputed for each reduction
  \end{itemize}
\end{frame}
